#### 1. 函数修改外部变量
void increase(int *num) {
    *num += 1; // 修改外部变量
}

int main() {
    int x = 5;
    increase(&x); // x变成6
}
#### 2. 数组操作
int arr[3] = {10,20,30};
int *p = arr; // 数组名就是首元素地址

printf("%d", *p);    // 10 → 第一个元素
printf("%d", *(p+1));// 20 → 第二个元素
#### 3. 字符串处理
char *str = "Hello";
while(*str != '\0') {
    putchar(*str); // 逐个输出字符
    str++;
}
#### 4. 动态内存分配
int *p = (int*)malloc(sizeof(int)*5); // 申请5个int空间
*p = 10;      // 第一个元素
*(p+1) = 20;  // 第二个元素
free(p);      // 释放内存
